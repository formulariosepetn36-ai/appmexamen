--   #3  Quién crea la mesa: aquí se MANTIENE RB-20 de ellos (la crea el primero
--       en anotarse; la fecha sale de es_convocatoria.fecha_mesa).
--   #4  Ventana de inscripción: mantiene ELLOS (una por convocatoria,
--       es_convocatoria.fecha_inicio_inscripcion / fecha_fin_inscripcion).
--   #10 Estados de inscripción: se mantiene 'inscripto' | 'anulado' de ellos
--       (sin etapa de aprobación); lo nuestro queda como trazabilidad (S5).
--   #17 Alcance: 3 aplicaciones (landing + SIGME web + SIGME-M).
--   #18 Nombres de las tecnicaturas: el seed usa los NOMBRES CORTOS del CSV y
--       deja anotada la unificación pendiente (ver bloque H).
--
-- CONVENCIONES APLICADAS (analisis.txt sección 1, sin recortes):
--   * id uuid PRIMARY KEY DEFAULT gen_random_uuid()
--   * created_at / updated_at / deleted_at con timestamptz (soft delete)
--   * función trg_set_updated_at() + CREATE TRIGGER en TODA tabla con updated_at
--     (excepciones declaradas: las junctions y aud_evento)
--   * índice parcial idx_{tabla}_activa ON {tabla}(id) WHERE deleted_at IS NULL
--   * ON DELETE explícito en CADA FK: CASCADE (junction/detalle),
--     RESTRICT (referencias que no se pierden), SET NULL (referencia opcional)
--   * nombres: uq_* (únicos), idx_* (índices), ck_* (checks); el prefijo de área
--     NO se repite dentro del nombre del índice o del check
--   * tablas junction: PK compuesta sobre las dos FK, sin id ni timestamps (1.9)
--   * varchar(n) con límite de negocio, nunca text sin tope (1.10)
--   * COMMENT ON en las columnas que no se explican solas
-- ==============================================================================

SET client_encoding = 'UTF8';

-- ------------------------------------------------------------------------------
-- 0. Auditoría de tabla — se crea UNA vez y la usan todas las tablas (1.6).
--    DEFAULT now() solo corre en INSERT: sin este trigger, updated_at queda
--    congelado en el valor del alta para siempre.
-- ------------------------------------------------------------------------------
CREATE OR REPLACE FUNCTION trg_set_updated_at() RETURNS trigger AS $$
BEGIN
  NEW.updated_at := now();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TABLE pe_documento_tipo (
    id            uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    descripcion   varchar(50) NOT NULL UNIQUE,   -- DNI, LC, LE, Pasaporte
    orden         int,
    activo        boolean NOT NULL DEFAULT true,
    created_at    timestamptz NOT NULL DEFAULT now(),
    updated_at    timestamptz,
    deleted_at    timestamptz
);
-- [CORREGIDO de analisis.txt] La v1 lo llamaba "pe_documentotipo" (sin guion)
-- mientras que la FK lo referenciaba como "pe_documentotipos" (plural). Esa FK
-- NO EXISTÍA y el CREATE TABLE fallaba.

CREATE TRIGGER trg_pe_documento_tipo_updated_at
  BEFORE UPDATE ON pe_documento_tipo
  FOR EACH ROW EXECUTE FUNCTION trg_set_updated_at();

CREATE INDEX idx_pe_documento_tipo_activa
  ON pe_documento_tipo(id) WHERE deleted_at IS NULL;

-- ------------------------------------------------------------------------------
-- A.2  pe_persona   (analisis.txt 3.2)
-- ------------------------------------------------------------------------------
CREATE TABLE pe_persona (
    id                 uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    documento_tipo_id  uuid NOT NULL REFERENCES pe_documento_tipo(id)
                                    ON DELETE RESTRICT,
    documento          varchar(20)  NOT NULL,
    apellido           varchar(100) NOT NULL,
    nombre             varchar(100) NOT NULL,
    razon_social       varchar(150),              -- nullable: persona física no tiene
    fecha_nacimiento   date,
    sexo               varchar(1),                -- nullable
    direccion          varchar(150),
    nacionalidad       varchar(50) NOT NULL DEFAULT 'Argentino',
    created_at         timestamptz NOT NULL DEFAULT now(),
    updated_at         timestamptz,
    deleted_at         timestamptz
);
-- [CORREGIDO de analisis.txt] La v1 marcaba NOT NULL a razon_social,
-- fecha_nacimiento, sexo, direccion y nacionalidad. Un esquema compartido con la
-- landing no puede exigir datos que esa app nunca pide: obligaba a insertar
-- basura.
--
-- Un D.N.I. = una persona (RB-01 nuestro; "no chocan" en comparativa §10).
-- Índice único PARCIAL (1.7): una persona borrada no bloquea re-insertar el
-- mismo documento.
CREATE UNIQUE INDEX uq_persona_documento
    ON pe_persona(documento_tipo_id, documento) WHERE deleted_at IS NULL;
CREATE INDEX idx_persona_apellido_nombre ON pe_persona(apellido, nombre);
CREATE INDEX idx_persona_activa ON pe_persona(id) WHERE deleted_at IS NULL;

CREATE TRIGGER trg_pe_persona_updated_at
  BEFORE UPDATE ON pe_persona
  FOR EACH ROW EXECUTE FUNCTION trg_set_updated_at();

CREATE TABLE us_sesion (
    id                  uuid PRIMARY KEY DEFAULT gen_random_uuid(),  -- sid
    usuario_id          uuid NOT NULL REFERENCES us_usuario(id)
                                     ON DELETE CASCADE,
    modulo_id           uuid NOT NULL REFERENCES us_modulo(id)
                                     ON DELETE RESTRICT,
    refresh_token_hash  varchar(64) NOT NULL UNIQUE,  -- SHA-256 hex, NUNCA crudo
    familia_id          uuid NOT NULL,                -- para reuse detection
    estado              varchar(12) NOT NULL DEFAULT 'activo'
                        CHECK (estado IN ('activo','revocado')),
    rotated_at          timestamptz,
    ip                  inet,
    user_agent          varchar(300),
    expires_at          timestamptz NOT NULL,
    created_at          timestamptz NOT NULL DEFAULT now(),
    -- [ADAPTACIÓN] analisis.txt 3.12 dejó updated_at / deleted_at fuera con la
    -- nota "[CORREGIDO] ... los tenía sin uso". Acá se los conserva porque la
    -- convención 1.4 pide el trío de auditoría "en TODA tabla" y ésta era la
    -- única tabla base que se desviaba: updated_at registra la rotación o la
    -- revocación y deleted_at archiva la sesión sin perder trazabilidad. El
    -- ciclo de vida sigue gobernando por `estado`.
    updated_at          timestamptz,
    deleted_at          timestamptz
);
CREATE INDEX idx_sesion_usuario ON us_sesion(usuario_id, estado);
CREATE INDEX idx_sesion_familia ON us_sesion(familia_id);
CREATE INDEX idx_sesion_expira  ON us_sesion(expires_at);
CREATE INDEX idx_sesion_activa  ON us_sesion(id) WHERE deleted_at IS NULL;

-- [CORREGIDO de analisis.txt] La v1 guardaba token y refreshToken en texto

CREATE TRIGGER trg_us_sesion_updated_at
  BEFORE UPDATE ON us_sesion
  FOR EACH ROW EXECUTE FUNCTION trg_set_updated_at();

-- [S2] La TABLA es de ellos, con su columna limite_materias intacta. La SEMILLA
--      es NUESTRA (docs/diseno.md D7): Regular, Previa, Completar Estudio,
--      Equivalencia — los 4 valores del impreso de la escuela. NO se siembra
--      'Libre' (bloque H.3).
--
-- limite_materias: IMPLEMENTADO el 2026-09-29 a pedido del usuario (cierra el
-- PD-02 de analisis.txt): Previa = 3, Regular = 8, y NULL = sin límite para
-- Completar Estudio y Equivalencia. Se valida SOLO al inscribirse (RB-03 y
-- RB-08c de ellos), NUNCA el día de la mesa (RB-17). El conteo es por
-- CONVOCATORIA, igual que uq_inscripcion_unica (convocatoria × alumno × materia).
--
-- OJO: es DISTINTO de RB-13 nuestro (máximo 6 FILAS por permiso, porque el
-- formulario A4 tiene 6 renglones). RB-13 se valida aparte, al GENERAR el
-- permiso (es_permiso_examen + es_permiso_inscripcion), nunca en esta tabla.
-- Ambos se sostienen: si Regular permitiera 8 materias y el impreso tiene 6
-- filas, el alumno necesita DOS permisos — exactamente lo que dice RB-13.
COMMENT ON TABLE es_condicion IS

--   nombre_alumno, documento, fecha_nacimiento, orientacion, nombre_tutor, email
--
-- O-1 RESUELTA — CONSENTIMIENTO FUERA DEL SISTEMA:
--   La preinscripción online es sólo una manifestación de interés. El
--   consentimiento real se recoge DESPUÉS, de forma física, en un formulario
--   en papel que se le entrega al interesado. Eso es un proceso EXTERNO y no
--   lo contempla este sistema.
--   Por eso NO existe columna `consentimiento` ni `consentimiento_en`.
--
--   Reparto de responsabilidades Ley 25.326:
--     · art. 6  aviso previo (para qué se usan los datos, derechos)
--                -> REQUISITO HEREDADO A LA LANDING (no es del backend)
--     · art. 5  consentimiento -> proceso físico externo, fuera del sistema
--     · art. 9  medidas de seguridad técnicas y organizativas -> BACKEND (doc 09)
--     · art. 14 acceso en 10 días corridos   -> BACKEND (doc 09)
--     · art. 16 rectificación/supresión en 5 días hábiles -> BACKEND (doc 09)
--
-- O-2 RESUELTA — TODOS los campos del payload son OBLIGATORIOS (así se
--   impusieron): nombre_alumno, documento, fecha_nacimiento, orientacion,
--   nombre_tutor, email. Los 6 son NOT NULL, incluido `nombre_tutor`

--   (ver E.2). El backend toma `anio_lectivo` de la ventana vigente; el
--   cliente NO lo envía.
--
-- O-4 RESUELTA — REPETIR EL FORMULARIO NO ES ERROR:
--   El usuario puede reenviar 2, 5 o 100 veces (cambia de orientación, se
--   equivoca, insiste). El backend NO devuelve 409 por documento repetido.
--   Es un UPSERT sobre la identidad (anio_lectivo, documento):

--     · Ya existía  -> UPDATE de los campos mutables, responde 200 OK.
--   Campos IDENTITARIOS (no cambian la identidad): anio_lectivo, documento.
--   Campos MUTABLES (se sobrescriben en cada reenvío):

--   La VERIFICACIÓN de los datos antes de enviar (resumen + doble confirmación

-- La landing es ANÓNIMA: cualquiera completa el formulario, sin roles.

-- ------------------------------------------------------------------------------
-- F.4  us_password_reset   (S6) — nuestro RB-19 + RB-07 + verificación de correo
-- ------------------------------------------------------------------------------
-- analisis.txt lo deja como PD-04 PENDIENTE ("¿Se requiere reseteo de
-- contraseña por email? Implica tabla adicional us_password_reset").
-- Nosotros lo traemos RESUELTO (comparativa §12): código de 6 dígitos, un solo
-- uso, 15 minutos, 5 intentos -> bloqueo de 15 min, error genérico anti-
-- enumeración. Sirve para dos propósitos: recuperar acceso y verificar correo.
CREATE TABLE us_password_reset (
    id           uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    usuario_id   uuid NOT NULL REFERENCES us_usuario(id) ON DELETE CASCADE,
    proposito    varchar(20) NOT NULL
                 CHECK (proposito IN ('reset_password','verify_email')),
    codigo_hash  varchar(128) NOT NULL,   -- hash del código de 6 dígitos, NUNCA en claro
    enviado_a    varchar(150) NOT NULL,   -- correo destino
    intentos     int NOT NULL DEFAULT 0 CHECK (intentos >= 0),  -- máx 5 -> bloqueo 15 min
    expira_en    timestamptz NOT NULL,    -- 15 minutos
    usado_en     timestamptz,
    created_at   timestamptz NOT NULL DEFAULT now(),
    -- [ADAPTACIÓN] convención 1.4: trío de auditoría completo, aunque el
    -- artefacto sea de un solo uso. updated_at registra el último intento y
    -- deleted_at archiva códigos vencidos sin perder la trazabilidad.
    updated_at   timestamptz,
    deleted_at   timestamptz
);
-- RB-19 / RB-07: un SOLO código activo por usuario y propósito. Reenviar
-- invalida el anterior (se lo marca usado y queda fuera de este índice).
CREATE UNIQUE INDEX uq_password_reset_activo
  ON us_password_reset(usuario_id, proposito) WHERE usado_en IS NULL;
CREATE INDEX idx_password_reset_usuario ON us_password_reset(usuario_id);
CREATE INDEX idx_password_reset_expira  ON us_password_reset(expira_en);
CREATE INDEX idx_password_reset_activa  ON us_password_reset(id)
    WHERE deleted_at IS NULL;

COMMENT ON TABLE  us_password_reset IS
  'Código de recuperación de acceso y de verificación de correo (RB-07/RB-19; cierra el PD-04 de analisis.txt).';
COMMENT ON COLUMN us_password_reset.codigo_hash IS
  'Hash del código de 6 dígitos; el código jamás se guarda en claro.';
COMMENT ON COLUMN us_password_reset.enviado_a IS
  'Correo destino (el de la ficha del alumno).';
COMMENT ON COLUMN us_password_reset.intentos IS
  'Intentos de ingreso del código: 5 fallidos -> bloqueo temporal de 15 minutos.';
COMMENT ON COLUMN us_password_reset.expira_en IS
  'Vence 15 minutos después de enviado.';
COMMENT ON COLUMN us_password_reset.usado_en IS
  'No nulo = código consumido; además lo saca del índice único de códigos activos.';

CREATE TRIGGER trg_us_password_reset_updated_at
  BEFORE UPDATE ON us_password_reset
  FOR EACH ROW EXECUTE FUNCTION trg_set_updated_at();

-- ------------------------------------------------------------------------------
-- F.5  aud_evento   (S6) — nuestro RB-10 + PD-05 de ellos. APPEND-ONLY.
-- ------------------------------------------------------------------------------
-- analisis.txt lo deja como PD-05 PENDIENTE ("¿Se necesita bitácora de
-- auditoría (quién modificó qué mesa)? Si sí: tabla aud_evento con snapshot
-- JSONB"). Nosotros ya la teníamos resuelta: RB-10, "toda acción del admin
-- queda registrada".
--
-- SIN updated_at / deleted_at y SIN trigger: es una bitácora de SOLO
-- ESCRITURA. Modificar o borrar una fila de auditoría anularía su valor de
-- prueba; por eso no se le aplica el trío de la convención 1.4 (excepción
-- declarada junto a las junctions).
CREATE TABLE aud_evento (
    id               uuid PRIMARY KEY DEFAULT gen_random_uuid(),
    actor_usuario_id uuid REFERENCES us_usuario(id) ON DELETE SET NULL,
    accion           varchar(60) NOT NULL,
    entidad          varchar(60) NOT NULL,
    entidad_id       uuid,
    antes            jsonb,
    despues          jsonb,
    created_at       timestamptz NOT NULL DEFAULT now(),
    ck_aud_evento_accion  CHECK (btrim(accion)   <> ''),
    ck_aud_evento_entidad CHECK (btrim(entidad)  <> ''),
    ck_aud_evento_antes   CHECK (antes   IS NULL OR jsonb_typeof(antes)   = 'object'),
    ck_aud_evento_despues CHECK (despues IS NULL OR jsonb_typeof(despues) = 'object')
);
CREATE INDEX idx_evento_entidad ON aud_evento(entidad, entidad_id);
CREATE INDEX idx_evento_created ON aud_evento(created_at);

COMMENT ON TABLE  aud_evento IS
  'Bitácora append-only de toda acción del admin (RB-10 nuestro / PD-05 de ellos).';
COMMENT ON COLUMN aud_evento.actor_usuario_id IS
  'Autor de la acción; SET NULL si el usuario se borra (la acción queda registrada).';
COMMENT ON COLUMN aud_evento.accion  IS
  'Acción, ej.: inscripcion.anular, permiso.regenerar.';
COMMENT ON COLUMN aud_evento.entidad IS
  'Entidad afectada, ej.: es_mesa_inscripcion, es_permiso_examen.';
COMMENT ON COLUMN aud_evento.antes   IS
  'Estado previo (jsonb) de la entidad editada.';
COMMENT ON COLUMN aud_evento.despues IS
  'Estado nuevo (jsonb) de la entidad editada.';

-- RB-22: filtro de "mesa con vida". Usar en TODO listado e impresión.
-- Una mesa queda con 0 alumnos si todos anularon; entonces se ignora.
CREATE VIEW vw_mesa_con_alumnos AS
SELECT m.*
FROM es_mesa_examen m
WHERE m.deleted_at IS NULL
  AND EXISTS (
      SELECT 1 FROM es_mesa_inscripcion i
      WHERE i.mesa_examen_id = m.id
        AND i.estado = 'inscripto'
        AND i.deleted_at IS NULL
  );
